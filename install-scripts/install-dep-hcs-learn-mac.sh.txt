#!/usr/bin/env bash
#
# setup.sh - Dev environment bootstrap for macOS and Linux
# Installs: Git, Node.js (+ npm), Visual Studio Code
# Then clones: https://github.com/shaveenbagelooTHA/learn-hcs-topic.git
#
# Usage: chmod +x setup.sh && ./setup.sh
#
set -euo pipefail

REPO_URL="https://github.com/shaveenbagelooTHA/learn-hcs-topic.git"
REPO_DIR="learn-hcs-topic"

info()  { printf "\033[1;34m[INFO]\033[0m  %s\n" "$1"; }
ok()    { printf "\033[1;32m[OK]\033[0m    %s\n" "$1"; }
warn()  { printf "\033[1;33m[WARN]\033[0m  %s\n" "$1"; }
error() { printf "\033[1;31m[ERROR]\033[0m %s\n" "$1"; }

OS="$(uname -s)"

# ---------------------------------------------------------------
# macOS
# ---------------------------------------------------------------
install_macos() {
    info "Detected macOS"

    # Homebrew
    if ! command -v brew >/dev/null 2>&1; then
        info "Installing Homebrew..."
        /bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
        # Add brew to PATH for this session (Apple Silicon vs Intel)
        if [ -x /opt/homebrew/bin/brew ]; then
            eval "$(/opt/homebrew/bin/brew shellenv)"
        elif [ -x /usr/local/bin/brew ]; then
            eval "$(/usr/local/bin/brew shellenv)"
        fi
    else
        ok "Homebrew already installed"
    fi

    # Git
    if ! command -v git >/dev/null 2>&1; then
        info "Installing Git..."
        brew install git
    else
        ok "Git already installed ($(git --version))"
    fi

    # Node.js + npm
    if ! command -v node >/dev/null 2>&1; then
        info "Installing Node.js (LTS)..."
        brew install node
    else
        ok "Node already installed ($(node --version), npm $(npm --version))"
    fi

    # VS Code
    if ! command -v code >/dev/null 2>&1 && [ ! -d "/Applications/Visual Studio Code.app" ]; then
        info "Installing Visual Studio Code..."
        brew install --cask visual-studio-code
    else
        ok "VS Code already installed"
    fi
}

# ---------------------------------------------------------------
# Linux (Debian/Ubuntu via apt, Fedora/RHEL via dnf)
# ---------------------------------------------------------------
install_linux() {
    info "Detected Linux"

    if command -v apt-get >/dev/null 2>&1; then
        PKG="apt"
    elif command -v dnf >/dev/null 2>&1; then
        PKG="dnf"
    else
        error "Unsupported Linux distro (need apt or dnf). Install git/node/code manually."
        exit 1
    fi

    SUDO=""
    if [ "$(id -u)" -ne 0 ]; then
        SUDO="sudo"
    fi

    if [ "$PKG" = "apt" ]; then
        info "Updating apt package lists..."
        $SUDO apt-get update -y

        # Git
        if ! command -v git >/dev/null 2>&1; then
            info "Installing Git..."
            $SUDO apt-get install -y git
        else
            ok "Git already installed ($(git --version))"
        fi

        # Node.js LTS via NodeSource (distro repos are often very old)
        if ! command -v node >/dev/null 2>&1; then
            info "Installing Node.js LTS via NodeSource..."
            $SUDO apt-get install -y curl ca-certificates
            curl -fsSL https://deb.nodesource.com/setup_lts.x | $SUDO -E bash -
            $SUDO apt-get install -y nodejs
        else
            ok "Node already installed ($(node --version), npm $(npm --version))"
        fi

        # VS Code via Microsoft repo
        if ! command -v code >/dev/null 2>&1; then
            info "Installing Visual Studio Code..."
            $SUDO apt-get install -y wget gpg apt-transport-https
            wget -qO- https://packages.microsoft.com/keys/microsoft.asc | gpg --dearmor > /tmp/packages.microsoft.gpg
            $SUDO install -D -o root -g root -m 644 /tmp/packages.microsoft.gpg /etc/apt/keyrings/packages.microsoft.gpg
            echo "deb [arch=amd64,arm64,armhf signed-by=/etc/apt/keyrings/packages.microsoft.gpg] https://packages.microsoft.com/repos/code stable main" | $SUDO tee /etc/apt/sources.list.d/vscode.list > /dev/null
            rm -f /tmp/packages.microsoft.gpg
            $SUDO apt-get update -y
            $SUDO apt-get install -y code
        else
            ok "VS Code already installed"
        fi

    elif [ "$PKG" = "dnf" ]; then
        # Git
        if ! command -v git >/dev/null 2>&1; then
            info "Installing Git..."
            $SUDO dnf install -y git
        else
            ok "Git already installed ($(git --version))"
        fi

        # Node.js
        if ! command -v node >/dev/null 2>&1; then
            info "Installing Node.js..."
            $SUDO dnf install -y nodejs npm
        else
            ok "Node already installed ($(node --version), npm $(npm --version))"
        fi

        # VS Code via Microsoft repo
        if ! command -v code >/dev/null 2>&1; then
            info "Installing Visual Studio Code..."
            $SUDO rpm --import https://packages.microsoft.com/keys/microsoft.asc
            printf "[code]\nname=Visual Studio Code\nbaseurl=https://packages.microsoft.com/yumrepos/vscode\nenabled=1\ngpgcheck=1\ngpgkey=https://packages.microsoft.com/keys/microsoft.asc\n" | $SUDO tee /etc/yum.repos.d/vscode.repo > /dev/null
            $SUDO dnf install -y code
        else
            ok "VS Code already installed"
        fi
    fi
}

# ---------------------------------------------------------------
# Clone the repo
# ---------------------------------------------------------------
clone_repo() {
    if [ -d "$REPO_DIR" ]; then
        warn "Directory '$REPO_DIR' already exists - skipping clone."
        warn "Run 'cd $REPO_DIR && git pull' to update it instead."
    else
        info "Cloning $REPO_URL ..."
        git clone "$REPO_URL"
        ok "Repo cloned into ./$REPO_DIR"
    fi
}

# ---------------------------------------------------------------
# Main
# ---------------------------------------------------------------
case "$OS" in
    Darwin) install_macos ;;
    Linux)  install_linux ;;
    *)
        error "Unsupported OS: $OS (use setup.ps1 on Windows)"
        exit 1
        ;;
esac

clone_repo

echo
ok "All done! Summary:"
echo "  git:  $(git --version 2>/dev/null || echo 'not found')"
echo "  node: $(node --version 2>/dev/null || echo 'not found')"
echo "  npm:  $(npm --version 2>/dev/null || echo 'not found')"
echo "  code: $(code --version 2>/dev/null | head -n1 || echo 'installed (open from Applications)')"
echo
echo "Next steps:"
echo "  cd $REPO_DIR"
echo "  npm install"
echo "  code ."
